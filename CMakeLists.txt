cmake_minimum_required(VERSION 3.20)
project(SekiroCraftPassthrough VERSION 0.1.0 LANGUAGES C CXX ASM)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(NOT WIN32 OR NOT CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "Windows x64 is required")
endif()
set(SC_DEPENDENCY_ROOT "${CMAKE_CURRENT_SOURCE_DIR}/third_party" CACHE PATH "Pinned MinHook and ImGui source root")
find_package(JNI REQUIRED)
add_subdirectory("${SC_DEPENDENCY_ROOT}/minhook" "${CMAKE_BINARY_DIR}/minhook")
add_library(sekirobridge-jni SHARED bridge/jni.cpp)
target_include_directories(sekirobridge-jni PRIVATE ${JNI_INCLUDE_DIRS})
target_compile_definitions(sekirobridge-jni PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
set_target_properties(sekirobridge-jni PROPERTIES PREFIX "" RUNTIME_OUTPUT_DIRECTORY "${CMAKE_CURRENT_SOURCE_DIR}/build/passthrough-native")
add_library(dinput8 SHARED src/proxy.cpp src/host_mod.cpp src/movement_hook.cpp src/movement_stub.S src/dinput8.def
    "${SC_DEPENDENCY_ROOT}/imgui/imgui.cpp" "${SC_DEPENDENCY_ROOT}/imgui/imgui_draw.cpp"
    "${SC_DEPENDENCY_ROOT}/imgui/imgui_widgets.cpp" "${SC_DEPENDENCY_ROOT}/imgui/imgui_tables.cpp"
    "${SC_DEPENDENCY_ROOT}/imgui/backends/imgui_impl_dx11.cpp" "${SC_DEPENDENCY_ROOT}/imgui/backends/imgui_impl_win32.cpp")
target_include_directories(dinput8 PRIVATE include "${SC_DEPENDENCY_ROOT}/imgui")
target_compile_definitions(dinput8 PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(dinput8 PRIVATE minhook d3d11 dxgi d3dcompiler bcrypt user32 gdi32 imm32 dwmapi dxguid)
set_target_properties(dinput8 PROPERTIES PREFIX "")
enable_testing()
add_executable(bridge_tests tests/bridge_tests.cpp)
add_test(NAME ipc COMMAND bridge_tests)
add_executable(compositor_tests tests/compositor_tests.cpp)
target_include_directories(compositor_tests PRIVATE include)
target_compile_definitions(compositor_tests PRIVATE NOMINMAX WIN32_LEAN_AND_MEAN)
target_link_libraries(compositor_tests PRIVATE d3d11 dxgi d3dcompiler)
add_test(NAME compositor COMMAND compositor_tests)
if(MINGW)
    foreach(target dinput8 sekirobridge-jni bridge_tests compositor_tests)
        target_link_options(${target} PRIVATE -static)
    endforeach()
endif()
